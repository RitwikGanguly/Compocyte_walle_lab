# Compocyte v2 — CPU requirements
#
# Reproduces the `compo` conda env used for the Compocyte v2_cpu benchmark
# (comparison/resource_comparison/train_compocyte_cpu.py).
# Tested with Python 3.12.14. pyproject.toml requires Python >= 3.12.
#
# System packages needed first (for pygraphviz / builds):
#   conda install -c conda-forge gcc gxx graphviz pygraphviz
#   (or: sudo apt-get install -y gcc g++ graphviz graphviz-dev)
#
# Install into a fresh env with:
#   conda create -n compocyte-cpu python=3.12 -y
#   conda activate compocyte-cpu
#   conda install -c conda-forge gcc gxx graphviz pygraphviz
#   pip install -r requirements-compocyte-cpu.txt
#   pip install -e .        # this repo (provides the Compocyte package itself)
#
# NOTE: torch==2.11.0 ships CUDA-13 wheels. On machines whose NVIDIA driver
# only supports CUDA <= 12.x, `torch.cuda.is_available()` is False and
# Compocyte transparently runs on CPU (expected for this file). If you have
# no GPU at all and want smaller wheels, replace the torch line with:
#   torch==2.11.0 --index-url https://download.pytorch.org/whl/cpu
